function updateCartTotals(data) {
    // 1. Оновлюємо список товарів кошика
    if (data.cart_html) {
        const cartList = document.querySelector('.cart-item-list');
        if (cartList) {
            cartList.innerHTML = data.cart_html;
        }
    }


    // 3. Оновлюємо загальну суму замовлення
    if (data.full_total_sum !== undefined) {
        // Підсумок "Сума замовлення"
        document.querySelectorAll('.total-val').forEach(el => {
            el.innerText = data.full_total_sum + " грн";
        });
        // Сума поруч з "Сума без доставки"
        document.querySelectorAll('.price-val').forEach(el => {
            el.innerText = data.cart_total_price + " грн";
        });
    }

    // 4. Оновлюємо лічильник у шапці
    const badge = document.getElementById('cart-counter');
    if (badge) {
        badge.innerText = data.cart_count;
        badge.style.display = data.cart_count > 0 ? 'flex' : 'none';
    }
}

function updateCartTotals(data) {
    // 1. Оновлюємо ВЕСЬ контент кошика (товари + підсумки + кнопку замовлення)
    if (data.cart_html) {
        const cartList = document.getElementById('cart-items-list');
        if (cartList) {
            cartList.innerHTML = data.cart_html;
            log('[Cart] HTML updated from server');
        }
    }

    // 2. Оновлюємо лічильник у шапці (бо він поза межами id="cart-list")
    if (typeof window.updateGlobalCartCounter === 'function') {
        window.updateGlobalCartCounter(data.cart_count);
    } else {
        const badge = document.getElementById('cart-counter');
        if (badge) {
            badge.innerText = data.cart_count;
            badge.style.display = data.cart_count > 0 ? 'flex' : 'none';
        }
    }

    // 3. Якщо якісь суми знаходяться ВІДОКРЕМЛЕНО від cart-list, 
    // тоді оновлюємо їх вручну (як ти і робила):
    if (data.full_total_sum !== undefined) {
        document.querySelectorAll('.total-val').forEach(el => {
            el.innerText = data.full_total_sum + " грн";
        });
    }
}
